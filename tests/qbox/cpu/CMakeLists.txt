set(_qbox_cpu_test_archs "")
foreach(arch IN LISTS QBOX_CPU_TEST_ARCHS)
    if(NOT arch MATCHES "^(aarch64|hexagon|riscv32)$")
        message(FATAL_ERROR "Unsupported QBOX_CPU_TEST_ARCHS entry: ${arch}")
    endif()
    if(arch IN_LIST _qbox_cpu_test_archs)
        message(FATAL_ERROR "Duplicate QBOX_CPU_TEST_ARCHS entry: ${arch}")
    endif()
    list(APPEND _qbox_cpu_test_archs "${arch}")
endforeach()

foreach(arch IN LISTS _qbox_cpu_test_archs)
    if(arch STREQUAL "aarch64")
        set(QBOX_CPU_TEST_ARCH_LIBS cpu_arm_cortexA53)
    elseif(arch STREQUAL "hexagon")
        set(QBOX_CPU_TEST_ARCH_LIBS qemu_cpu_hexagon hexagon_globalreg)
    else()
        set(QBOX_CPU_TEST_ARCH_LIBS cpu_riscv32 plic_sifive)
    endif()
    add_subdirectory("${arch}")
endforeach()
